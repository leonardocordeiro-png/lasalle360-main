    <script>
        // Simulação de interatividade
        document.getElementById('bookingForm').addEventListener('submit', function(e) {
            e.preventDefault();
            document.getElementById('confirmModal').classList.remove('hidden');
        });

        function closeModal() {
            document.getElementById('confirmModal').classList.add('hidden');
        }

        // Atualização dinâmica de disponibilidade
        document.getElementById('quantity').addEventListener('input', function(e) {
            const qty = parseInt(e.target.value) || 0;
            const available = 200 - 30; // Simulação
            const remaining = available - qty;
            
            if (remaining < 0) {
                e.target.classList.add('border-red-500');
                document.getElementById('maxAvailable').textContent = 'Quantidade excede disponível!';
                document.getElementById('maxAvailable').classList.remove('text-green-600');
                document.getElementById('maxAvailable').classList.add('text-red-600');
            } else {
                e.target.classList.remove('border-red-500');
                document.getElementById('maxAvailable').textContent = remaining;
                document.getElementById('maxAvailable').classList.remove('text-red-600');
                document.getElementById('maxAvailable').classList.add('text-green-600');
            }
        });

        // Animação de entrada dos cards
        const cards = document.querySelectorAll('.animate-slide-in');
        cards.forEach((card, index) => {
            card.style.animationDelay = `${index * 0.1}s`;
        });
    </script>
